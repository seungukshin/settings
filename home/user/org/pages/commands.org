* build debian packages for Linux kernel
#+begin_src sh
$ git clone https://github.com/torvalds/linux.git -b v6.18 --depth 1
$ cd linux

$ sudo apt build-dep linux linux-image-unsigned-$(uname -r)
$ sudo apt-get install build-essential bc kmod cpio flex libncurses5-dev libelf-dev libssl-dev dwarves bison gcc make exuberant-ctags gawk openssl dkms libudev-dev libpci-dev libiberty-dev autoconf llvm libdw-dev debhelper

$ cp /boot/config-6.8.0-88-generic .config
# set CONFIG_SYSTEM_TRUSTED_KEYS=""
# set CONFIG_SYSTEM_REVOCATION_KEYS=""
$ make olddefconfig

$ LOCALVERSION=-rx-zerocopy-2 make -j 32 bindeb-pkg
#+end_src
